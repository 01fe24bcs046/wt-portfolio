Node.js — JWT Auth Middleware & Protected Routes
const jwt = require("jsonwebtoken");

// JWT Middleware
const protect = (req, res, next) => {
  const token = req.headers.authorization?.split(" ")[1];
  if (!token) return res.status(401).json({ msg: "Not authorized" });
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch { res.status(401).json({ msg: "Token expired" }); }
};

// Role-based middleware
const requireRole = (...roles) => (req, res, next) => {
  if (!roles.includes(req.user.role))
    return res.status(403).json({ msg: "Access denied" });
  next();
};

// Protected route — only managers
router.post("/tasks", protect, requireRole("manager"), async (req, res) => {
  const task = await Task.create({ ...req.body, createdBy: req.user._id });
  res.status(201).json(task);
});

// Reusable TaskCard Component
function TaskCard({ task, onComplete }) {

    "In Progress": "#f59e0b",
    "Completed": "#22c55e",
    "Overdue": "#ef4444",
  };

  return (

        {task.status}
      </span>
      <p>Due: {new Date(task.deadline).toLocaleDateString()}</p>
      <button onClick={() => onComplete(task._id)}>
        Mark Complete
      </button>
    </div>
  );
}

// Parent component with state
function TaskList() {
  const [tasks, setTasks] = useState([]);

  const handleComplete = (id) => {
    setTasks(prev => prev.map(t =>
      t._id === id ? { ...t, status: "Completed" } : t
    ));
  };

  return (
    <div>
      {tasks.map(task => (
        <TaskCard key={task._id} task={task} onComplete={handleComplete} />
      ))}
    </div>
  );
}

import { useState, useEffect } from "react";
import { Navigate } from "react-router-dom";

// useEffect for data fetching
function Dashboard() {
  const [tasks, setTasks] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const load = async () => {
      const res = await fetch("/api/tasks", {
        headers: { Authorization: `Bearer ${localStorage.getItem("token")}` }
      });
      const data = await res.json();
      setTasks(data);
      setLoading(false);
    };
    load();
  }, []); // runs once on mount

  if (loading) return <p>Loading...</p>;
  return <TaskList tasks={tasks} />;
}

// Protected Route — only allow certain roles
function ProtectedRoute({ role, children }) {
  const user = JSON.parse(localStorage.getItem("user"));
  if (!user) return <Navigate to="/login" />;
  if (user.role !== role) return <Navigate to="/unauthorized" />;
  return children;
}

// Load tasks from localStorage on page load
let tasks = JSON.parse(localStorage.getItem("tasks")) || [];

function saveTasks() {
  localStorage.setItem("tasks", JSON.stringify(tasks));
}

function renderTasks(filter = "all") {
  const list = document.getElementById("task-list");
  list.innerHTML = "";

  const filtered = tasks.filter(t => {
    if (filter === "active") return !t.done;
    if (filter === "done")   return t.done;
    return true;
  });

  filtered.forEach(task => {
    const li = document.createElement("li");
    li.className = task.done ? "done" : "";
    li.innerHTML = `
      <span onclick="toggleTask(${task.id})">${task.text}</span>
      <button onclick="deleteTask(${task.id})">✕</button>
    `;
    list.appendChild(li);
  });
}

function addTask() {
  const input = document.getElementById("task-input");
  if (!input.value.trim()) return;
  tasks.push({ id: Date.now(), text: input.value.trim(), done: false });
  input.value = "";
  saveTasks();
  renderTasks();
}

function toggleTask(id) {
  tasks = tasks.map(t => t.id === id ? { ...t, done: !t.done } : t);
  saveTasks(); renderTasks();
}

document.addEventListener("DOMContentLoaded", () => renderTasks());

Assignment 3 — Express REST API with Auth
const router = require("express").Router();
const bcrypt = require("bcryptjs");
const jwt    = require("jsonwebtoken");
const User   = require("../models/User");

// POST /api/auth/register
router.post("/register", async (req, res) => {
  try {
    const { name, email, password } = req.body;
    const exists = await User.findOne({ email });
    if (exists) return res.status(400).json({ msg: "Email already registered" });

    const hash = await bcrypt.hash(password, 10);
    const user = await User.create({ name, email, password: hash });
    res.status(201).json({ msg: "User registered", id: user._id });
  } catch (err) {
    res.status(500).json({ msg: err.message });
  }
});

// POST /api/auth/login
router.post("/login", async (req, res) => {
  const { email, password } = req.body;
  const user = await User.findOne({ email });
  if (!user || !(await bcrypt.compare(password, user.password)))
    return res.status(401).json({ msg: "Invalid credentials" });

  const token = jwt.sign({ id: user._id }, process.env.JWT_SECRET, { expiresIn: "7d" });
  res.json({ token, user: { id: user._id, name: user.name } });
});

module.exports = router;

📖 Problem Statement

Organizations often face difficulties managing tasks, tracking employee progress, and maintaining communication between managers and employees. Manual methods lead to delayed work, poor coordination, lack of transparency, and inefficient workload management.

Objective: Build a role-based web application (Super Admin → Manager → Employee) automating task distribution, tracking, review, and feedback with full audit logging.

📋 No Accountability

💬 Communication Gaps

DevOps

Version control, CI/CD, and cloud deployment practices used in TaskBridge

🔧 What is DevOps?
DevOps is a set of practices that combines software development (Dev) and IT operations (Ops). It aims to shorten the development lifecycle, improve collaboration, and enable continuous delivery. For TaskBridge, we used Git for version control, GitHub for collaboration, and Vercel for deployment.
🐙

Git & GitHub — Version Control

We used Git for tracking every change in our code. The entire project is hosted on GitHub with separate repositories for frontend and backend. Branching strategies helped us work in parallel without conflicts.

Git — Common Commands Used
# Initialize & connect to GitHub
git init
git remote add origin https://github.com/victorykings2828-bot/WT-PORTFOLIO.git

# Daily workflow
git status                     # check what changed
git add .                      # stage all changes
git commit -m "feat: add dashboard"
git push origin main

# Branching
git checkout -b feature/login  # create new branch
git merge feature/login        # merge when done
git pull origin main           # get latest changes

# Useful commands
git log --oneline              # see commit history
git diff                       # see uncommitted changes
git stash                      # save work temporarily
🚀

Vercel — Deployment Platform

The TaskBridge frontend is deployed on Vercel. Every push to the main branch automatically triggers a new deployment — this is Continuous Deployment (CD). The live app is available at taskbridge-frontend-sandy.vercel.app.

🍃

MongoDB Atlas — Cloud Database

MongoDB Atlas provides a fully managed cloud database. We configured connection strings, IP whitelists, and database users — all DevOps tasks ensuring the database is secure, backed up, and always available.

🔐

Environment Variables & Security

Sensitive information (MongoDB URI, JWT secret, OAuth keys) is stored in .env files, never committed to GitHub. Vercel's environment variables panel stores production secrets securely.

.env — Environment Variables
# Backend .env
MONGO_URI=mongodb+srv://user:pass@cluster.mongodb.net/taskbridge
JWT_SECRET=your_super_secret_key_here
PORT=5000
GOOGLE_CLIENT_ID=xxx.apps.googleusercontent.com
EMAIL_USER=noreply@taskbridge.com

# Frontend .env
VITE_API_URL=https://api.taskbridge.com
VITE_GOOGLE_CLIENT_ID=xxx.apps.googleusercontent.com
